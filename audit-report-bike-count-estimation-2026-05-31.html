After (Suggested Fix)

    m.fit(X_train, y_train_h24[:, h],
          eval_set=[(X_val, y_val_h24[:, h])])
    return h, m, m.predict(X_val)

results = Parallel(n_jobs=-1)(
    delayed(_fit_one)(h) for h in range(24))
for h, model, preds in sorted(results):
